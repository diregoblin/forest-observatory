cmake_minimum_required(VERSION 3.20)
project(forest_hydrology LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(hydro_core STATIC hydro.cpp)
set_target_properties(hydro_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_options(hydro_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(hydro_checks checks.cpp)
target_link_libraries(hydro_checks PRIVATE hydro_core)
enable_testing()
add_test(NAME hydro_checks COMMAND hydro_checks)
option(FOREST_EXTENSION "Build Redot GDExtension" ON)
if(FOREST_EXTENSION)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/profile.json" "${CMAKE_CURRENT_SOURCE_DIR}/api/extension_api.json")
  set(GODOTCPP_GDEXTENSION_DIR "${CMAKE_CURRENT_SOURCE_DIR}/api" CACHE PATH "" FORCE)
  set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/profile.json" CACHE PATH "" FORCE)
  set(GODOTCPP_TARGET template_release CACHE STRING "" FORCE)
  add_subdirectory(deps/redot-cpp)
  add_library(forest_hydro SHARED extension.cpp)
  target_link_libraries(forest_hydro PRIVATE hydro_core godot-cpp)
  set_target_properties(forest_hydro PROPERTIES
    CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/lib")
endif()
